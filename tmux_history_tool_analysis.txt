    g = G[(r.Patient, r.Session, r.Section)]

    m = np.zeros(len(p), bool)
    for s, e in zip(pos.start_time, pos.stop_time):

    p_in = p[m].mean() if m.any() else float("nan")

          f"proc={len(p)/sf/60:5.1f} min anot={g.stop_time.max()/60:5.1f} min "
